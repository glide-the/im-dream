"""
Story Workspace shared types — backend contract.

This module defines the Python-side type contracts for story-workspace.
Frontend types are maintained separately in frontend/src/types/story-workspace/.

Naming convention: All types use StoryWorkspace* prefix.
"""

from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
from typing import List, Optional, Dict, Any, Generic, TypeVar


# =============================================================================
# Enums
# =============================================================================

class ReviewStatus(str, Enum):
    """Review status for stories, characters, and scenes."""
    PENDING = "pending"
    CONFIRMED = "confirmed"
    REJECTED = "rejected"


class ContentStatus(str, Enum):
    """Content publication status."""
    DRAFT = "draft"
    PUBLISHED = "published"
    ARCHIVED = "archived"


class StoryType(str, Enum):
    """Story type classification."""
    SHORT = "short"
    LONG = "long"
    SCRIPT = "script"
    OUTLINE = "outline"


class RoleType(str, Enum):
    """Character role type in a story."""
    PROTAGONIST = "protagonist"
    SUPPORTING = "supporting"
    EXTRA = "extra"


class BatchAction(str, Enum):
    """Batch review action types."""
    CONFIRM = "confirm"
    REJECT = "reject"
    ARCHIVE = "archive"


class ResourceType(str, Enum):
    """Resource types for batch operations."""
    STORY = "story"
    CHARACTER = "character"
    SCENE = "scene"


# =============================================================================
# Base Models
# =============================================================================

@dataclass
class StoryWorkspaceBase:
    """Base fields for all story-workspace entities."""
    id: str
    created_at: datetime
    updated_at: datetime


# =============================================================================
# Workspace Models
# =============================================================================

@dataclass
class StoryWorkspaceWorkspace:
    """Workspace entity."""
    id: str
    name: str
    owner_id: int
    settings: Dict[str, Any] = field(default_factory=dict)
    created_at: Optional[datetime] = None
    updated_at: Optional[datetime] = None


# =============================================================================
# Story Models
# =============================================================================

@dataclass
class StoryWorkspaceStory:
    """Story/Script entity generated by Agent."""
    id: str
    identifier: str
    title: str
    description: Optional[str] = None
    status: ContentStatus = ContentStatus.DRAFT
    review_status: ReviewStatus = ReviewStatus.PENDING
    type: StoryType = StoryType.SHORT
    content: Optional[str] = None
    author_id: int = 0
    workspace_id: str = ""
    character_count: int = 0
    scene_count: int = 0
    agent_generated: bool = True
    agent_session_id: Optional[str] = None
    review_notes: Optional[str] = None
    created_at: Optional[datetime] = None
    updated_at: Optional[datetime] = None
    confirmed_at: Optional[datetime] = None
    published_at: Optional[datetime] = None


@dataclass
class StoryWorkspaceStoryDetail(StoryWorkspaceStory):
    """Story detail with related characters and scenes."""
    characters: List["StoryWorkspaceCharacter"] = field(default_factory=list)
    scenes: List["StoryWorkspaceScene"] = field(default_factory=list)


# =============================================================================
# Character Models
# =============================================================================

@dataclass
class StoryWorkspaceCharacter:
    """Character entity generated by Agent."""
    id: str
    identifier: str
    name: str
    avatar_url: Optional[str] = None
    identity: Optional[str] = None
    personality: Optional[str] = None
    background: Optional[str] = None
    catchphrase: Optional[str] = None
    tags: List[str] = field(default_factory=list)
    notes: Optional[str] = None
    author_id: int = 0
    workspace_id: str = ""
    story_count: int = 0
    review_status: ReviewStatus = ReviewStatus.PENDING
    agent_generated: bool = True
    created_at: Optional[datetime] = None
    updated_at: Optional[datetime] = None


@dataclass
class StoryWorkspaceCharacterDetail(StoryWorkspaceCharacter):
    """Character detail with related stories."""
    stories: List[StoryWorkspaceStory] = field(default_factory=list)


# =============================================================================
# Scene Models
# =============================================================================

@dataclass
class StoryWorkspaceScene:
    """Scene entity generated by Agent."""
    id: str
    identifier: str
    name: str
    description: Optional[str] = None
    story_id: Optional[str] = None
    author_id: int = 0
    workspace_id: str = ""
    character_count: int = 0
    order_index: int = 0
    review_status: ReviewStatus = ReviewStatus.PENDING
    agent_generated: bool = True
    created_at: Optional[datetime] = None
    updated_at: Optional[datetime] = None


@dataclass
class StoryWorkspaceSceneDetail(StoryWorkspaceScene):
    """Scene detail with related characters and story."""
    characters: List[StoryWorkspaceCharacter] = field(default_factory=list)
    story: Optional[StoryWorkspaceStory] = None


# =============================================================================
# Association Models
# =============================================================================

@dataclass
class StoryWorkspaceStoryCharacter:
    """Story-Character association."""
    story_id: str
    character_id: str
    role_type: RoleType = RoleType.SUPPORTING
    created_at: Optional[datetime] = None


@dataclass
class StoryWorkspaceSceneCharacter:
    """Scene-Character association."""
    scene_id: str
    character_id: str
    created_at: Optional[datetime] = None


# =============================================================================
# API Request/Response Types
# =============================================================================

T = TypeVar('T')


@dataclass
class PaginatedResponse(Generic[T]):
    """Standard paginated API response."""
    data: List[T]
    pagination: "PaginationInfo"


@dataclass
class PaginationInfo:
    """Pagination metadata."""
    page: int
    per_page: int
    total: int
    total_pages: int


@dataclass
class StoryFilter:
    """Filter parameters for story list queries."""
    q: Optional[str] = None
    review_status: Optional[List[ReviewStatus]] = None
    status: Optional[List[ContentStatus]] = None
    type: Optional[List[StoryType]] = None


@dataclass
class CharacterFilter:
    """Filter parameters for character list queries."""
    q: Optional[str] = None
    review_status: Optional[List[ReviewStatus]] = None


@dataclass
class SceneFilter:
    """Filter parameters for scene list queries."""
    q: Optional[str] = None
    review_status: Optional[List[ReviewStatus]] = None
    story_id: Optional[str] = None


@dataclass
class BatchReviewRequest:
    """Request body for batch review operations."""
    action: BatchAction
    resource_type: ResourceType
    ids: List[str]
    review_notes: Optional[str] = None


@dataclass
class BatchReviewResponse:
    """Response for batch review operations."""
    success: bool
    action: BatchAction
    resource_type: ResourceType
    total_requested: int
    total_updated: int
    skipped_ids: List[str]
    updated_items: List[Dict[str, Any]]


@dataclass
class ReviewActionRequest:
    """Request body for single-item reject action."""
    review_notes: Optional[str] = None


@dataclass
class WorkspaceStats:
    """Workspace statistics for Dashboard."""
    pending_count: int
    confirmed_count: int
    total_count: int


@dataclass
class AgentOutputRequest:
    """Request body for Agent output storage (internal endpoint)."""
    agent_session_id: str
    user_id: int
    workspace_id: Optional[str] = None
    story: "AgentStoryOutput"


@dataclass
class AgentStoryOutput:
    """Agent-generated story content (minimum data contract)."""
    title: str
    description: Optional[str] = None
    type: Optional[str] = "short"
    content: Optional[str] = None
    characters: Optional[List["AgentCharacterOutput"]] = None
    scenes: Optional[List["AgentSceneOutput"]] = None


@dataclass
class AgentCharacterOutput:
    """Agent-generated character content."""
    name: str
    identity: Optional[str] = None
    personality: Optional[str] = None
    background: Optional[str] = None
    catchphrase: Optional[str] = None
    tags: Optional[List[str]] = None


@dataclass
class AgentSceneOutput:
    """Agent-generated scene content."""
    name: str
    description: Optional[str] = None
    order_index: Optional[int] = 0

NAMING_CONVENTIONS = {
